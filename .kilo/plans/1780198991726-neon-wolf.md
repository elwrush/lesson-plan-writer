## Task 2 — Component Extraction: `scripts/slides_builder.py`

### Purpose
Replace hand-written 8-12 line HTML fragments with Python function calls. An LLM calling `answer_row("1.", "Statement", "Answer", why="...")` is far less error-prone than writing the entire `<div class="a-row">...</div>` structure from scratch.

### Module: `scripts/slides_builder.py`

```python
"""HTML fragment builders for reveal.js slides.

Every function returns a string of well-formed HTML suitable for
insertion into <div class="slides">. All strings are raw HTML —
no templating engine, no escape gotchas. The caller composes them
into the splice file.

Design principles:
- Functions output fragment strings, not file-write or file-read
- No side effects — pure string builders
- Indentation uses 4-space for readability in output
- All text content is the caller's responsibility (no XSS concerns
  since we generate static files, not template user input)
"""

import html as _html


# ---------------------------------------------------------------------------
# Answer-list row (the most frequently repeated pattern)
# ---------------------------------------------------------------------------

def answer_row(
    num: str,           # "1." or "4."
    question: str,      # statement/question short text
    answer: str,        # "b — parent-teen relationships" or "False"
    fragment_index: int, # 1-based, each row gets a unique index
    *,
    is_correct: bool = True,
    why: str | None = None,  # transcript quote or explanation for WHY line
    show_icon: bool = True,
) -> str:
    """Generate one answer-list row with fragment fade-up reveal.

    Produces:
        <div class="a-row">
            <span class="a-num">N.</span>
            <span class="a-q">question text</span>
            <span class="fragment fade-up a-ans {a-cor|a-inc}"
                  data-fragment-index="N">
                <i class="fa-solid fa-check|fa-times" style="color:#fff;"></i>
                answer text
            </span>
            <span class="fragment fade-up" data-fragment-index="N"
                  style="width:100%; color:#ffdd00; font-size:0.95em; text-align:left;">
                WHY: transcript quote
            </span>
        </div>
    """
    cls = "a-cor" if is_correct else "a-inc"
    icon = "fa-check" if is_correct else "fa-times"
    icon_html = f'<i class="fa-solid {icon}" style="color:#fff;"></i> ' if show_icon else ""

    lines = []
    lines.append('        <div class="a-row">')
    lines.append(f'            <span class="a-num">{_html.escape(num)}</span>')
    lines.append(f'            <span class="a-q">{_html.escape(question)}</span>')
    lines.append(f'            <span class="fragment fade-up a-ans {cls}" data-fragment-index="{fragment_index}">{icon_html}{_html.escape(answer)}</span>')
    if why:
        lines.append(f'            <span class="fragment fade-up" data-fragment-index="{fragment_index}" style="width:100%; color:#ffdd00; font-size:0.95em; text-align:left;">WHY: {_html.escape(why)}</span>')
    lines.append('        </div>')
    return "\n".join(lines)


# ---------------------------------------------------------------------------
# Answer slide wrapper
# ---------------------------------------------------------------------------

def answer_slide(
    slide_id: str,
    heading: str,         # "Exercises 1–3 — Answers (1–3)"
    rows: list[str],      # pre-rendered answer_row() fragments
    *,
    notes: str | None = None,
) -> str:
    """Wrap answer rows in a full green-background answer slide."""
    parts = [
        f'<section id="{slide_id}" data-background="#1e7e34">',
        f'    <h2>{_html.escape(heading)}</h2>',
        '    <div class="answer-list">',
    ]
    parts.extend(rows)
    parts.append('    </div>')
    if notes:
        parts.append(f'    <aside class="notes">{_html.escape(notes)}</aside>')
    parts.append('</section>')
    return "\n".join(parts)


# ---------------------------------------------------------------------------
# Transition slide
# ---------------------------------------------------------------------------

def transition_slide(slide_id: str, heading: str) -> str:
    """Red-background phase-change signal. Heading only."""
    return (
        f'<section id="{slide_id}" data-background="#c0392b">\n'
        f'    <h2>{_html.escape(heading)}</h2>\n'
        f'</section>'
    )


# ---------------------------------------------------------------------------
# Task instruction slide
# ---------------------------------------------------------------------------

def task_slide(
    slide_id: str,
    heading: str,
    instruction: str,
    *,
    audio_src: str | None = None,
    timer: int | None = None,
    notes: str | None = None,
) -> str:
    """Task slide — brief student-facing instruction + audio OR timer.

    Raises ValueError if both audio_src and timer are set (per CRITICAL rule:
    never place a timer pill on a slide that plays audio).
    """
    if audio_src and timer:
        raise ValueError("task_slide: cannot have both audio_src and timer")

    attrs = f' id="{slide_id}" data-background="#1a1a2e"'
    if audio_src:
        attrs += f' data-audio-src="{audio_src}"'
    if timer is not None:
        attrs += f' data-timer="{timer}"'

    parts = [
        f'<section{attrs}>',
        f'    <h2>{_html.escape(heading)}</h2>',
        f'    <p>{_html.escape(instruction)}</p>',
    ]
    if notes:
        parts.append(f'    <aside class="notes">{_html.escape(notes)}</aside>')
    parts.append('</section>')
    return "\n".join(parts)


# ---------------------------------------------------------------------------
# Pedagogical strategy slide (non-auto-animate)
# ---------------------------------------------------------------------------

def pedagogical_slide(
    slide_id: str,
    content_html: str,  # raw HTML — caller builds <h2>, <p>, <ul> etc.
    *,
    notes: str | None = None,
) -> str:
    """Teal-background pedagogical slide wrapper."""
    parts = [
        f'<section id="{slide_id}" class="pedagogical" data-background="#1a6b5a" data-background-transition="none">',
        content_html,
    ]
    if notes:
        parts.append(f'    <aside class="notes">{_html.escape(notes)}</aside>')
    parts.append('</section>')
    return "\n".join(parts)


# ---------------------------------------------------------------------------
# Pedagogical intent comment
# ---------------------------------------------------------------------------

def intent_comment(
    intent: str,       # what student SEES happen
    feature: str,      # reveal.js feature + why alternatives fail
    principle: str,    # Mayer's cognitive principle
) -> str:
    """Mandatory pedagogical intent annotation block."""
    return (
        f'<!-- PEDAGOGICAL INTENT: {intent} -->\n'
        f'<!-- WHY THIS FEATURE: {feature} -->\n'
        f'<!-- COGNITIVE PRINCIPLE: {principle} -->'
    )


# ---------------------------------------------------------------------------
# Aside (speaker notes)
# ---------------------------------------------------------------------------

def aside_notes(text: str) -> str:
    """Speaker notes block."""
    return f'<aside class="notes">{_html.escape(text)}</aside>'


# ---------------------------------------------------------------------------
# Title slide
# ---------------------------------------------------------------------------

def title_slide(
    topic: str,
    cefr: str,           # "B1", "B2", etc.
    strap: str,           # derived from lesson objective
    *,
    bg_image: str | None = None,
    bg_opacity: str = "0.7",
    notes: str | None = None,
) -> str:
    """Title slide with logo, topic, CEFR badge, and strap."""
    attrs = f' id="slide-title"'
    if bg_image:
        attrs += f' data-background-image="{bg_image}" data-background-opacity="{bg_opacity}"'
    parts = [
        f'<section{attrs}>',
        '    <img src="assets/logo.png" class="title-logo" alt="Logo" />',
        f'    <h1>{_html.escape(topic)} <span class="cefr-badge {cefr}">{cefr}</span></h1>',
        f'    <p><em>{_html.escape(strap)}</em></p>',
    ]
    if notes:
        parts.append(f'    <aside class="notes">{_html.escape(notes)}</aside>')
    parts.append('</section>')
    return "\n".join(parts)


# ---------------------------------------------------------------------------
# Objective slide
# ---------------------------------------------------------------------------

def objective_slide(
    slide_id: str,
    objectives: list[str],
    *,
    footer: str | None = None,
    notes: str | None = None,
) -> str:
    """Objective slide with 3 outcomes."""
    parts = [
        f'<section id="{slide_id}">',
        '    <h2>Here\'s what you\'ll be able to do</h2>',
        '    <ul>',
    ]
    for obj in objectives:
        parts.append(f'        <li>{_html.escape(obj)}</li>')
    parts.append('    </ul>')
    if footer:
        parts.append(f'    <p><em>{_html.escape(footer)}</em></p>')
    if notes:
        parts.append(f'    <aside class="notes">{_html.escape(notes)}</aside>')
    parts.append('</section>')
    return "\n".join(parts)


# ---------------------------------------------------------------------------
# End slide
# ---------------------------------------------------------------------------

def end_slide(topic: str, cefr: str) -> str:
    """Dark blue-gray exit slide."""
    return (
        f'<section id="slide-end" data-background="#2c3e50">\n'
        f'    <h2>Thank you</h2>\n'
        f'    <p><em>{_html.escape(topic)}</em> | {_html.escape(cefr)}</p>\n'
        f'</section>'
    )
```

## Task 3 — Auto-Animate Helper Functions

### Observed Patterns (from codebase audit)

**Pattern 1: Keyword underline reveal** (M3-LISTENING-WB, M2-LISTENING-WB, 28-5-PRON, Reading-REX-03)
- Entry slide: keywords wrapped in `<span data-id="kN" style="border-bottom:2px solid transparent;">`
- Reveal slide: same spans, `border-bottom:2px solid #4fc3f7` (or `#ff8a65`)
- Both slides share `data-auto-animate` + matching `data-auto-animate-id`

**Pattern 2: Grammar rule highlight** (M3-Grammar-1, 5 rule pairs)
- Entry slide: sentence with plain spans that have `data-id` attributes
- Reveal slide: same spans with `style="color:#ffdd00; font-weight:bold;"` or `style="text-decoration: line-through; color: rgba(255,255,255,0.35);"`

**Pattern 3: S/V/O annotation** (M2-Grammar-2-Subjects-Verbs-and-Objects)
- Entry slide: 3 spans with transparent borders/invisible superscripts
- Reveal slide: 3 spans with colored borders (blue #4fc3f7, orange #ff8a65, green #aed581) + visible S/V/O superscripts + legend row
- Most complex pattern — 3 simultaneous visual transformations

```python
# ---------------------------------------------------------------------------
# Auto-animate: keyword underline reveal (Pattern 1)
# ---------------------------------------------------------------------------

def auto_animate_underline_pair(
    auto_animate_id: str,
    slide_id_prefix: str,   # "slide-strategy-detail" → entry="...-entry", reveal="...-reveal"
    title: str,
    intro_text: str | None,  # lead-in paragraph before the keyword list
    items: list[str],        # sentences with [[keyword]] markers to underline
    *,
    underline_color: str = "#4fc3f7",
    reveal_extra_html: str | None = None,  # content added only on reveal slide (e.g. challenge tip)
    reveal_notes: str | None = None,
) -> tuple[str, str]:
    """Generate an auto-animate pair (entry + reveal) for keyword underline reveals.

    Each item in `items` may contain `[[keyword text]]` markers.
    The function splits each item into text segments and generates matching
    <span data-id="kN"> elements with transparent (entry) or colored (reveal)
    border-bottom.

    Returns (entry_html, reveal_html) as two complete <section> strings.
    """
    import re

    def render_items(color: str) -> list[str]:
        """Build the rendered list items for one slide."""
        rendered = []
        for i, item in enumerate(items):
            parts = re.split(r"\[\[(.+?)\]\]", item)
            spans = []
            for j, part in enumerate(parts):
                if j % 2 == 0:
                    # plain text
                    if part:
                        spans.append(_html.escape(part))
                else:
                    # keyword — wrap in span with data-id
                    kid = f"k{i}_{j // 2}"
                    if color == "transparent":
                        spans.append(
                            f'<span data-id="{kid}" style="border-bottom:2px solid transparent;">{_html.escape(part)}</span>'
                        )
                    else:
                        spans.append(
                            f'<span data-id="{kid}" style="border-bottom:2px solid {color};">{_html.escape(part)}</span>'
                        )
            rendered.append(f"        <li>{''.join(spans)}</li>")
        return rendered

    # Entry slide
    entry_id = f"{slide_id_prefix}-entry"
    entry_title_id = f".title_{auto_animate_id}"
    lines_entry = [
        f'<section data-auto-animate data-auto-animate-id="{auto_animate_id}" id="{entry_id}" class="pedagogical" data-background="#1a6b5a" data-background-transition="none">',
        f'    <h2 data-id="{entry_title_id}">{_html.escape(title)}</h2>',
    ]
    if intro_text:
        lines_entry.append(f'    <p style="font-size:0.85em;">{_html.escape(intro_text)}</p>')
    lines_entry.append('    <ul style="font-size:0.8em;">')
    lines_entry.extend(render_items("transparent"))
    lines_entry.append('    </ul>')
    lines_entry.append('</section>')

    # Reveal slide
    reveal_id = f"{slide_id_prefix}-reveal"
    lines_reveal = [
        f'<section data-auto-animate data-auto-animate-id="{auto_animate_id}" id="{reveal_id}" class="pedagogical" data-background="#1a6b5a" data-background-transition="none">',
        f'    <h2 data-id="{entry_title_id}">{_html.escape(title)}</h2>',
    ]
    if intro_text:
        lines_reveal.append(f'    <p style="font-size:0.85em;">{_html.escape(intro_text)}</p>')
    lines_reveal.append('    <ul style="font-size:0.8em;">')
    lines_reveal.extend(render_items(underline_color))
    lines_reveal.append('    </ul>')
    if reveal_extra_html:
        lines_reveal.append(reveal_extra_html)
    if reveal_notes:
        lines_reveal.append(f'    <aside class="notes">{_html.escape(reveal_notes)}</aside>')
    lines_reveal.append('</section>')

    return "\n".join(lines_entry), "\n".join(lines_reveal)


# ---------------------------------------------------------------------------
# Auto-animate: grammar rule highlight (Pattern 2)
# ---------------------------------------------------------------------------

def auto_animate_highlight_pair(
    auto_animate_id: str,
    slide_id_prefix: str,
    title: str,
    entry_parts: list[dict],   # [{"html": "plain text", ...}, {"html": "...", "data_id": "r2-dummy"}, ...]
    reveal_parts: list[dict],  # same structure, but with reveal styles on specific data_ids
    *,
    rule_text: str | None = None,
    explanation_html: str | None = None,
    reveal_notes: str | None = None,
) -> tuple[str, str]:
    """Generate auto-animate pair for grammar rule transformations.

    entry_parts and reveal_parts are lists of dicts:
        {"html": "text content", "data_id": "optional-data-id", "style": "optional-inline-style"}

    Elements with matching data_id animate between entry and reveal states.
    This is Pattern 2 (grammar rules — M3-Grammar-1).
    """
    def render_parts(parts: list[dict]) -> list[str]:
        result = []
        for p in parts:
            html_text = p.get("html", "")
            data_id = p.get("data_id")
            style = p.get("style", "")
            if data_id:
                if style:
                    result.append(f'<span data-id="{data_id}" style="{style}">{html_text}</span>')
                else:
                    result.append(f'<span data-id="{data_id}">{html_text}</span>')
            else:
                if style:
                    result.append(f'<span style="{style}">{html_text}</span>')
                else:
                    result.append(html_text)
        return result

    entry_id = f"{slide_id_prefix}-entry"
    reveal_id = f"{slide_id_prefix}-reveal"

    entry_lines = [
        f'<section data-auto-animate data-auto-animate-id="{auto_animate_id}" id="{entry_id}" class="pedagogical" data-background="#1a6b5a" data-background-transition="none">',
        f'    <h2>{_html.escape(title)}</h2>',
    ]
    if rule_text:
        entry_lines.append(f'    <p><u><strong>{_html.escape(rule_text)}</strong></u></p>')
    entry_lines.append(f'    <p data-id="{auto_animate_id}-sentence"><em>{"".join(render_parts(entry_parts))}</em></p>')
    entry_lines.append('</section>')

    reveal_lines = [
        f'<section data-auto-animate data-auto-animate-id="{auto_animate_id}" id="{reveal_id}" class="pedagogical" data-background="#1a6b5a" data-background-transition="none">',
        f'    <h2>{_html.escape(title)}</h2>',
    ]
    if rule_text:
        reveal_lines.append(f'    <p><u><strong>{_html.escape(rule_text)}</strong></u></p>')
    reveal_lines.append(f'    <p data-id="{auto_animate_id}-sentence"><em>{"".join(render_parts(reveal_parts))}</em></p>')
    if explanation_html:
        reveal_lines.append(f'    {explanation_html}')
    if reveal_notes:
        reveal_lines.append(f'    <aside class="notes">{_html.escape(reveal_notes)}</aside>')
    reveal_lines.append('</section>')

    return "\n".join(entry_lines), "\n".join(reveal_lines)


# ---------------------------------------------------------------------------
# Auto-animate: S/V/O annotation pair (Pattern 3)
# ---------------------------------------------------------------------------

def auto_animate_svo_pair(
    auto_animate_id: str,
    slide_id_prefix: str,
    title: str,
    subject_word: str,
    verb_word: str,
    object_word: str,
    *,
    subject_color: str = "#4fc3f7",
    verb_color: str = "#ff8a65",
    object_color: str = "#aed581",
    reveal_extra_html: str | None = None,
    reveal_notes: str | None = None,
) -> tuple[str, str]:
    """Generate S/V/O annotation auto-animate pair (Pattern 3).

    Entry slide: plain sentence with invisible superscripts and transparent borders.
    Reveal slide: colored borders + visible S/V/O labels + legend row.
    Used for grammar lead-in demonstrations (M2-Grammar-2 pattern).
    """
    entry_id = f"{slide_id_prefix}-entry"
    reveal_id = f"{slide_id_prefix}-annotated"

    entry = (
        f'<section id="{entry_id}" data-auto-animate data-auto-animate-id="{auto_animate_id}" data-background="#1a1a2e">\n'
        f'    <h2 data-id="title">{_html.escape(title)}</h2>\n'
        f'    <p style="font-size: 1.3em; margin-top: 1em;">\n'
        f'        <span data-id="subject" style="border-bottom: 2px solid transparent;">\n'
        f'            <sup style="color: transparent; font-size: 0.5em;">S </sup>{_html.escape(subject_word)}\n'
        f'        </span>\n'
        f'        <span data-id="verb" style="border-bottom: 2px solid transparent;">\n'
        f'            <sup style="color: transparent; font-size: 0.5em;">V </sup>{_html.escape(verb_word)}\n'
        f'        </span>\n'
        f'        <span data-id="object" style="border: 2px solid transparent;">\n'
        f'            <sup style="color: transparent; font-size: 0.5em;">O </sup>{_html.escape(object_word)}\n'
        f'        </span>\n'
        f'    </p>\n'
        f'</section>'
    )

    reveal_lines = [
        f'<section id="{reveal_id}" data-auto-animate data-auto-animate-id="{auto_animate_id}" data-background="#1a1a2e">',
        f'    <h2 data-id="title">{_html.escape(title)}</h2>',
        f'    <p style="font-size: 1.3em; margin-top: 1em;">',
        f'        <span data-id="subject" style="border-bottom: 3px solid {subject_color};">',
        f'            <sup style="color: {subject_color}; font-size: 0.5em;">S </sup>{_html.escape(subject_word)}',
        f'        </span>',
        f'        <span data-id="verb" style="border-bottom: 3px solid {verb_color}; box-shadow: 0 5px 0 0 {verb_color};">',
        f'            <sup style="color: {verb_color}; font-size: 0.5em;">V </sup>{_html.escape(verb_word)}',
        f'        </span>',
        f'        <span data-id="object" style="border: 2px solid {object_color}; padding: 0 4px; border-radius: 4px;">',
        f'            <sup style="color: {object_color}; font-size: 0.5em;">O </sup>{_html.escape(object_word)}',
        f'        </span>',
        f'    </p>',
        f'    <p style="font-size: 0.85em; margin-top: 0.3em; color: #ccc;">',
        f'        <span style="color: {subject_color};"><strong>Subject</strong></span>',
        f'        &nbsp;&middot;&nbsp;',
        f'        <span style="color: {verb_color};"><strong>Verb</strong></span>',
        f'        &nbsp;&middot;&nbsp;',
        f'        <span style="color: {object_color};"><strong>Object</strong></span>',
        f'    </p>',
    ]
    if reveal_extra_html:
        reveal_lines.append(reveal_extra_html)
    if reveal_notes:
        reveal_lines.append(f'    <aside class="notes">{_html.escape(reveal_notes)}</aside>')
    reveal_lines.append('</section>')

    return entry, "\n".join(reveal_lines)
```
